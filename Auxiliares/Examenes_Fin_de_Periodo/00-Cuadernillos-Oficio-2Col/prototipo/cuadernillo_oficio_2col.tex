  \def\endhead{}\def\endfirsthead{}\def\endfoot{}\def\endlastfoot{}%
  \def\bottomrule{}%
  \begin{adjustbox}{max width=\linewidth}\begin{tabular}{#2}}{%
  \origbottomrule\end{tabular}\end{adjustbox}\par\smallskip}
