		\end{scope}

			\clip (F) circle (\R);
			% Rellenamos la parte del círculo de Mora que está en esa intersección.
			\fill[white] (Mo) circle (\R);
		\end{scope}

		\node[setlabel, anchor=south] at ($(F) + (0, \R)$) {Jugo de Fresa};
