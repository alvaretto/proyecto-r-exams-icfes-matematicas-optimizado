%% Aquí las listas no llevan sangría: si una opción o una pregunta EMPIEZA con una figura o una tabla, su etiqueta
%% («b)», «5.») se queda sola en su renglón y la figura va debajo, a todo el ancho. Sin esto la etiqueta y la figura
%% comparten renglón y lo desbordan (medido: Overfull de 9,8 a 15,3 pt en las opciones gráficas).
\makeatletter
\newcommand{\etiquetasola}{\ifvmode\if@inlabel\leavevmode\par\nobreak\fi\fi}
\makeatother
\let\includegraphicsoriginal\includegraphics
\renewcommand{\includegraphics}[2][]{\etiquetasola
  \adjustbox{max width=\linewidth,\ifvmode center\fi}{\includegraphicsoriginal[#1]{#2}}}
\usepackage{etoolbox}
\AtBeginEnvironment{figure}{\renewcommand{\caption}[2][]{}\centering}

%% tablas de pandoc (longtable) como tabular limitado al ancho de la columna (82 mm)
\usepackage{longtable}
\let\origbottomrule\bottomrule
\renewenvironment{longtable}[2][]{%
  \etiquetasola\par\smallskip\centering
  \def\endhead{}\def\endfirsthead{}\def\endfoot{}\def\endlastfoot{}%
  \def\bottomrule{}%
  \begin{adjustbox}{max width=\linewidth}\begin{tabular}{#2}}{%
  \origbottomrule\end{tabular}\end{adjustbox}\par\smallskip}

%% bloques de código de pandoc (verbatim): parten el renglón en los espacios en lugar de salirse de la página (medido:
