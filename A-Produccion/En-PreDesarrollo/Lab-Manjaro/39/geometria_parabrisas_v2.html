import matplotlib.pyplot as plt
import numpy as np

def crear_grafico_parabrisa(largo=', largo_base, ', ancho=', ancho_base, ', longitud_plumilla=', longitud_plumilla_base, ', opcion="principal"):

        # Arco para la plumilla
        theta = np.linspace(0, np.pi/2, 100)
        ax.plot(x_origen + radio * np.cos(theta), y_origen + radio * np.sin(theta), "k--")
        
        # Líneas radiales
        for i in range(5):
            angulo = (np.pi/2) * i / 4
            x_final = x_origen + radio * np.cos(angulo)
            y_final = y_origen + radio * np.sin(angulo)
            ax.plot([x_origen, x_final], [y_origen, y_final], "k--")
    
    # Configuración según la opción
    if opcion == "principal":
        # Diagrama principal con etiqueta "Plumilla"
        dibujar_plumilla(1, 0, 2.5)
        dibujar_plumilla(3.75, 0, 2.5)
